### Smart App Store Detection
```javascript
class AppStoreDetection {
  constructor() {
    this.init();
  }
  
  init() {
    this.setupAppStoreLinks();
    this.trackDownloadClicks();
  }
  
  setupAppStoreLinks() {
    const iosButton = document.getElementById('ios-download');
    const androidButton = document.getElementById('android-download');
    
    const userAgent = navigator.userAgent.toLowerCase();
    const isIOS = /iphone|ipad|ipod/.test(userAgent);
    const isAndroid = /android/.test(userAgent);
    
    // Smart routing based on device
    if (isIOS) {
      iosButton.classList.add('primary-cta');
      androidButton.classList.add('secondary-cta');
    } else if (isAndroid) {
      androidButton.classList.add('primary-cta');
      iosButton.classList.add('secondary-cta');
    }
    
    // Add click handlers
    iosButton.addEventListener('click', this.handleIOSClick.bind(this));
    androidButton.addEventListener('click', this.handleAndroidClick.bind(this));
  }
  
  handleIOSClick(event) {
    event.preventDefault();
    // Track conversion event
    this.trackConversion('ios_download_click');
    // Redirect to App Store
    window.open('https://apps.apple.com/app/magerly', '_blank');
  }
  
  handleAndroidClick(event) {
    event.preventDefault();
    // Track conversion event
    this.trackConversion('android_download_click');
    // Redirect to Google Play
    window.open('https://play.google.com/store/apps/details?id=com.magerly', '_blank');
  }
  
  trackConversion(event) {
    // Analytics tracking implementation
    if (typeof gtag !== 'undefined') {
      gtag('event', 'conversion', {
        'send_to': 'AW-XXXXXXXXX/XXXXXXX',
        'event_category': 'app_download',
        'event_label': event
      });
    }
  }
}
```
